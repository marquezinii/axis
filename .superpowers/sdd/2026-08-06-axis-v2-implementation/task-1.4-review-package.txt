+impl ClaudeProvider {
+    pub fn new() -> Self {
+        Self {
+            client: reqwest::Client::new(),
+        }
+    }
+
+    fn api_key() -> Result<String, ProviderError> {
+        std::env::var("ANTHROPIC_API_KEY")
+            .map_err(|_| ProviderError::AuthExpired)
+    }
+
+    fn build_body(
+        session: &SessionContext,
+        message: &str,
+        history: &[ChatMessage],
+        tools: &[ToolDefinition],
+    ) -> Value {
+        let mut messages: Vec<Value> = history
+            .iter()
+            .map(|m| {
+                serde_json::json!({
+                    "role": m.role,
+                    "content": m.content
+                })
+            })
+            .collect();
+
+        // Add the new user message
+        messages.push(serde_json::json!({
+            "role": "user",
+            "content": message
+        }));
+
+        let mut body = serde_json::json!({
+            "model": "claude-sonnet-4-20250514",
+            "max_tokens": 4096,
+            "messages": messages,
+            "stream": true,
+        });
+
+        if !tools.is_empty() {
+            let anthropic_tools: Vec<Value> = tools
+                .iter()
+                .map(|t| {
+                    serde_json::json!({
+                        "name": t.name,
+                        "description": t.description,
+                        "input_schema": t.input_schema,
+                    })
+                })
+                .collect();
+            body["tools"] = serde_json::json!(anthropic_tools);
+        }
+
+        body
+    }

+impl ChatProvider for ClaudeProvider {
+    fn id(&self) -> &'static str {
+        "claude"
+    }
+
+    fn display_name(&self) -> &'static str {
+        "Claude"
+    }
+
+    async fn send_message(
+        &self,
+        session: &SessionContext,
+        message: &str,
+        history: &[ChatMessage],
+        tools: &[ToolDefinition],
+        tx: tokio::sync::mpsc::Sender<StreamEvent>,
+    ) -> Result<(), ProviderError> {
+        let api_key = Self::api_key()?;
+        let body = Self::build_body(session, message, history, tools);
+
+        let response = self
+            .client
+            .post("https://api.anthropic.com/v1/messages")
+            .header("x-api-key", &api_key)
+            .header("anthropic-version", "2023-06-01")
+            .header("content-type", "application/json")
+            .json(&body)
+            .send()
+            .await
+            .map_err(|e| {
+                if e.is_timeout() {
+                    ProviderError::ProviderUnavailable
+                } else {
+                    ProviderError::Unknown(e.to_string())
+                }
+            })?;
+
+        if !response.status().is_success() {
+            let status = response.status().as_u16();
+            let body_text = response.text().await.unwrap_or_default();
+            return Err(match status {
+                401 => ProviderError::AuthExpired,
+                429 => ProviderError::RateLimited,
+                529 => ProviderError::ProviderUnavailable,
+                _ => ProviderError::Unknown(format!("HTTP {}: {}", status, body_text)),
+            });
+        }
+
+        let mut stream = response.bytes_stream();
+        let mut buffer = String::new();
+        let mut input_tokens: u64 = 0;
+        let mut output_tokens: u64 = 0;
+
+        while let Some(chunk_result) = stream.next().await {
+            let chunk = chunk_result.map_err(|e| ProviderError::Unknown(e.to_string()))?;
+            let text = String::from_utf8_lossy(&chunk);
+            buffer.push_str(&text);
+
+            // Process complete SSE events
+            while let Some(event_end) = buffer.find("\n\n") {
+                let event_text = buffer[..event_end].to_string();
+                buffer = buffer[event_end + 2..].to_string();
+
+                for line in event_text.lines() {
+                    if line.is_empty() || !line.starts_with("data: ") {
+                        continue;
+                    }
+                    let data = &line[6..];
+
+                    let event: Value = match serde_json::from_str(data) {
+                        Ok(v) => v,
+                        Err(_) => continue,
+                    };
+
+                    let event_type = event["type"].as_str().unwrap_or("");
+
+                    match event_type {
+                        "message_start" => {
+                            if let Some(usage) = event["message"]["usage"].as_object() {
+                                input_tokens = usage.get("input_tokens").and_then(|v| v.as_u64()).unwrap_or(0);
+                            }
+                        }
+                        "content_block_delta" => {
+                            if let Some(delta) = event["delta"]["text"].as_str() {
+                                let _ = tx.send(StreamEvent::TextDelta {
+                                    delta: delta.to_string(),
+                                }).await;
+                            }
+                        }
+                        "message_delta" => {
+                            if let Some(usage) = event["usage"]["output_tokens"].as_u64() {
+                                output_tokens = usage;
+                            }
+                        }
+                        "message_stop" => {
+                            let _ = tx.send(StreamEvent::SessionCompleted {
+                                input_tokens,
+                                output_tokens,
+                            }).await;
+                        }
+                        "error" => {
+                            let msg = event["error"]["message"]
+                                .as_str()
+                                .unwrap_or("Unknown Claude error");
+                            let _ = tx.send(StreamEvent::SessionError {
+                                code: "UNKNOWN".into(),
+                                message: msg.to_string(),
+                            }).await;
+                        }
+                        _ => {}
+                    }
+                }
+            }
+        }
+
+        Ok(())
+    }
